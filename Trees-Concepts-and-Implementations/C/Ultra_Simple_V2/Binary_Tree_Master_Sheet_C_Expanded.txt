BINARY TREE MASTER SHEET - C
Simple logic, traversal categories, and frequently asked problems

============================================================
MASTER MAP
============================================================

PREORDER DFS:
Current -> Left -> Right
Use when current node works before children.
Examples: Path Sum, Good Nodes, Invert Tree, Serialize.

INORDER DFS:
Left -> Current -> Right
Use when BST sorted order is important.
Examples: Kth Smallest, Recover BST.

POSTORDER DFS:
Left -> Right -> Current
Use when parent needs child answers.
Examples: Height, Diameter, Balanced Tree, LCA, Maximum Path Sum.

BFS:
Process one complete level at a time.
Examples: Level Order, Zigzag, Views, Width, Minimum Depth.

DOUBLE DFS:
Compare two nodes or two trees together.
Examples: Same Tree, Symmetric Tree, Subtree.

BST SINGLE PATH:
Use BST ordering and visit only one path.
Examples: Search, Insert, Delete, Floor, Ceil, LCA in BST.

int max2(int first, int second)
{
    if (first > second)
        return first;

    return second;
}

TreeNode *create_node(int value)
{
    TreeNode *node = malloc(sizeof(TreeNode));

    if (node == NULL)
        return NULL;

    node->val = value;
    node->left = NULL;
    node->right = NULL;

    return node;
}

int is_leaf(TreeNode *node)
{
    if (node == NULL)
        return 0;

    if (node->left != NULL)
        return 0;

    if (node->right != NULL)
        return 0;

    return 1;
}

2. HEIGHT - POSTORDER
============================================================

Problem:
Find the longest root-to-leaf height.

Logic:
Ask left.
Ask right.
Take the bigger one.
Add 1 for current node.

int tree_height(TreeNode *root)
{
    if (root == NULL)
        return 0;

    int left_height = tree_height(root->left);

    int right_height = tree_height(root->right);

    int bigger_height = max2(left_height, right_height);

    return 1 + bigger_height;
}

============================================================
3. DIAMETER - POSTORDER
============================================================

Problem:
Find the longest path between any two nodes.

Logic change from height:
Path through current node = left height + right height.

int diameter_answer = 0;

int height_for_diameter(TreeNode *root)
{
    if (root == NULL)
        return 0;

    int left_height = height_for_diameter(root->left);

    int right_height = height_for_diameter(root->right);

    int path_through_current = left_height + right_height;

    if (path_through_current > diameter_answer)
        diameter_answer = path_through_current;

    return 1 + max2(left_height, right_height);
}

int diameter(TreeNode *root)
{
    diameter_answer = 0;

    height_for_diameter(root);

    return diameter_answer;
}

4. BALANCED TREE - POSTORDER
============================================================

Problem:
At every node, left and right heights differ by at most 1.

Logic:
Return -1 immediately when a subtree is unbalanced.

int balanced_height(TreeNode *root)
{
    if (root == NULL)
        return 0;

    int left_height = balanced_height(root->left);

    if (left_height == -1)
        return -1;

    int right_height = balanced_height(root->right);

    if (right_height == -1)
        return -1;

    int difference = left_height - right_height;

    if (difference < 0)
        difference = -difference;

    if (difference > 1)
        return -1;

    return 1 + max2(left_height, right_height);
}

int is_balanced(TreeNode *root)
{
    int result = balanced_height(root);

    if (result == -1)
        return 0;

    return 1;
}

int same_tree(TreeNode *first, TreeNode *second)
{
    if (first == NULL && second == NULL)
        return 1;

    if (first == NULL || second == NULL)
        return 0;

    if (first->val != second->val)
        return 0;

    int left_same = same_tree(
        first->left,
        second->left);

    int right_same = same_tree(
        first->right,
        second->right);

    return left_same && right_same;
}

6. SYMMETRIC TREE - DOUBLE DFS / MIRROR

int mirror(TreeNode *left_node, TreeNode *right_node)
{
    if (left_node == NULL && right_node == NULL)
        return 1;

    if (left_node == NULL || right_node == NULL)
        return 0;

    if (left_node->val != right_node->val)
        return 0;

    int outside_same = mirror(
        left_node->left,
        right_node->right);

    int inside_same = mirror(
        left_node->right,
        right_node->left);

    return outside_same && inside_same;
}

int is_symmetric(TreeNode *root)
{
    if (root == NULL)
        return 1;

    return mirror(root->left, root->right);
}

int is_subtree(TreeNode *root, TreeNode *subroot)
{
    if (subroot == NULL)
        return 1;

    if (root == NULL)
        return 0;

    if (same_tree(root, subroot))
        return 1;

    int found_left = is_subtree(root->left, subroot);

    if (found_left)
        return 1;

    int found_right = is_subtree(root->right, subroot);

    return found_right;
}

                rear++;

9. ZIGZAG - BFS
============================================================

Only logic change:
Store one level.
Reverse alternate levels.

void zigzag(TreeNode *root)
{
    if (root == NULL)
        return;

    TreeNode *queue[MAXN];
    int level_values[MAXN];

    int front = 0;
    int rear = 0;
    int left_to_right = 1;

    queue[rear] = root;
    rear++;

    while (front < rear) {
        int level_size = rear - front;

        for (int index = 0; index < level_size; index++) {
            TreeNode *node = queue[front];
            front++;

            level_values[index] = node->val;

            if (node->left != NULL) {
                queue[rear] = node->left;
                rear++;
            }

            if (node->right != NULL) {
                queue[rear] = node->right;
                rear++;
            }
        }

        if (left_to_right) {
            for (int index = 0; index < level_size; index++)
                printf("%d ", level_values[index]);
        } else {
            for (int index = level_size - 1; index >= 0; index--)
                printf("%d ", level_values[index]);
        }

        printf("\n");

        left_to_right = !left_to_right;
    }
}

int has_path_sum(TreeNode *root, int target)
{
    if (root == NULL)
        return 0;

    int remaining = target - root->val;

    if (is_leaf(root))
        return remaining == 0;

    int left_result = has_path_sum(
        root->left,
        remaining);

    if (left_result)
        return 1;

    int right_result = has_path_sum(
        root->right,
        remaining);

    return right_result;
}

int good_nodes_helper(TreeNode *root, int maximum_seen)
{
    if (root == NULL)
        return 0;

    int current_is_good = 0;

    if (root->val >= maximum_seen)
        current_is_good = 1;

    int new_maximum = maximum_seen;

    if (root->val > new_maximum)
        new_maximum = root->val;

    int left_count = good_nodes_helper(
        root->left,
        new_maximum);

    int right_count = good_nodes_helper(
        root->right,
        new_maximum);

    return current_is_good + left_count + right_count;
}

int count_good_nodes(TreeNode *root)
{
    if (root == NULL)
        return 0;

    return good_nodes_helper(root, root->val);
}

17. MAXIMUM WIDTH - BFS WITH POSITIONS
============================================================

long long maximum_width(TreeNode *root)
{
    if (root == NULL)
        return 0;

    QueueItem queue[MAXN];

    int front = 0;
    int rear = 0;

    queue[rear].node = root;
    queue[rear].position = 0;
    rear++;

    long long best_width = 0;

    while (front < rear) {
        int level_size = rear - front;

        long long base_position = queue[front].position;

        long long first = 0;
        long long last = 0;

        for (int index = 0; index < level_size; index++) {
            QueueItem item = queue[front];
            front++;

            TreeNode *node = item.node;

            long long position =
                item.position - base_position;

            if (index == 0)
                first = position;

            if (index == level_size - 1)
                last = position;

            if (node->left != NULL) {
                queue[rear].node = node->left;
                queue[rear].position =
                    2 * position + 1;
                rear++;
            }

            if (node->right != NULL) {
                queue[rear].node = node->right;
                queue[rear].position =
                    2 * position + 2;
                rear++;
            }
        }

        long long current_width =
            last - first + 1;

        if (current_width > best_width)
            best_width = current_width;
    }

    return best_width;
}

int leftmost_height(TreeNode *root)
{
    int height = 0;

    while (root != NULL) {
        height++;
        root = root->left;
    }

    return height;
}

int rightmost_height(TreeNode *root)
{
    int height = 0;

    while (root != NULL) {
        height++;
        root = root->right;
    }

    return height;
}

int count_complete_nodes(TreeNode *root)
{
    if (root == NULL)
        return 0;

    int left_height = leftmost_height(root);

    int right_height = rightmost_height(root);

    if (left_height == right_height)
        return (1 << left_height) - 1;

    int left_count = count_complete_nodes(root->left);

    int right_count = count_complete_nodes(root->right);

    return 1 + left_count + right_count;
}

TreeNode *insert_bst(TreeNode *root, int value)
{

        return create_node(value);

    if (value < root->val) {
        TreeNode *new_left = insert_bst(
            root->left,

        root->left = new_left;
    } else if (value > root->val) {
        TreeNode *new_right = insert_bst(
            root->right,
            value);

        root->right = new_right;
    }

    return root;
}

21. FLOOR AND CEIL - BST SINGLE PATH
============================================================

int bst_floor(TreeNode *root, int target, int *found)
{
    int answer = 0;

    *found = 0;

    while (root != NULL) {
        if (root->val == target) {
            *found = 1;
            return root->val;
        }

        if (root->val < target) {
            answer = root->val;
            *found = 1;
            root = root->right;
        } else {
            root = root->left;
        }
    }

    return answer;
}

int bst_ceil(TreeNode *root, int target, int *found)
{
    int answer = 0;

    *found = 0;

    while (root != NULL) {
        if (root->val == target) {
            *found = 1;
            return root->val;
        }

        if (root->val > target) {
            answer = root->val;
            *found = 1;
            root = root->left;
        } else {
            root = root->right;
        }
    }

    return answer;
}

23. KTH SMALLEST - INORDER

void kth_helper(
    TreeNode *root,
    int k,
    int *count,
    int *answer,
    int *found)
{
    if (root == NULL)
        return;

    if (*found)
        return;

    kth_helper(
        root->left,
        k,
        count,
        answer,
        found);

    if (*found)
        return;

    (*count)++;

    if (*count == k) {
        *answer = root->val;
        *found = 1;
        return;
    }

    kth_helper(
        root->right,
        k,
        count,
        answer,
        found);
}

int kth_smallest(TreeNode *root, int k, int *found)
{
    int count = 0;
    int answer = 0;

    *found = 0;

    kth_helper(
        root,
        k,
        &count,
        &answer,
        found);

    return answer;
}

TreeNode *lca_bst(
    TreeNode *root,
    TreeNode *first,
    TreeNode *second)
{
    while (root != NULL) {
        if (first->val < root->val &&
            second->val < root->val) {
            root = root->left;
            continue;
        }

        if (first->val > root->val &&
            second->val > root->val) {
            root = root->right;
            continue;
        }

        return root;
    }

    return NULL;
}

25. BST DELETE

TreeNode *minimum_node(TreeNode *root)
{
    while (root != NULL && root->left != NULL)
        root = root->left;

    return root;
}

TreeNode *delete_bst(TreeNode *root, int value)
{
    if (root == NULL)
        return NULL;

    if (value < root->val) {
        root->left = delete_bst(
            root->left,
            value);

        return root;
    }

    if (value > root->val) {
        root->right = delete_bst(
            root->right,
            value);

        return root;
    }

    if (root->left == NULL) {
        TreeNode *right_child = root->right;

        free(root);

        return right_child;
    }

    if (root->right == NULL) {
        TreeNode *left_child = root->left;

        free(root);

        return left_child;
    }

    TreeNode *successor = minimum_node(root->right);

    root->val = successor->val;

    root->right = delete_bst(
        root->right,
        successor->val);

    return root;
}

int maximum_path_answer = INT_MIN;

int maximum_gain(TreeNode *root)
{
    if (root == NULL)
        return 0;

    int left_gain = maximum_gain(root->left);

    int right_gain = maximum_gain(root->right);

    if (left_gain < 0)
        left_gain = 0;

    if (right_gain < 0)
        right_gain = 0;

    int complete_path =
        root->val + left_gain + right_gain;

    if (complete_path > maximum_path_answer)
        maximum_path_answer = complete_path;

    int better_child = max2(
        left_gain,
        right_gain);

    return root->val + better_child;
}

int maximum_path_sum(TreeNode *root)
{
    maximum_path_answer = INT_MIN;

    maximum_gain(root);

    return maximum_path_answer;
}

27. MINIMUM DEPTH - BFS
============================================================

int minimum_depth(TreeNode *root)
{
    if (root == NULL)
        return 0;

    TreeNode *queue[MAXN];
    int depth_queue[MAXN];

    int front = 0;
    int rear = 0;

    queue[rear] = root;
    depth_queue[rear] = 1;
    rear++;

    while (front < rear) {
        TreeNode *node = queue[front];
        int depth = depth_queue[front];
        front++;

        if (is_leaf(node))
            return depth;

        if (node->left != NULL) {
            queue[rear] = node->left;
            depth_queue[rear] = depth + 1;
            rear++;
        }

        if (node->right != NULL) {
            queue[rear] = node->right;
            depth_queue[rear] = depth + 1;
            rear++;
        }
    }

    return 0;
}

28. CHILDREN SUM PROPERTY
============================================================

void change_children_sum(TreeNode *root)
{
    if (root == NULL)
        return;

    int child_sum = 0;

    if (root->left != NULL)
        child_sum += root->left->val;

    if (root->right != NULL)
        child_sum += root->right->val;

    if (child_sum >= root->val) {
        root->val = child_sum;
    } else {
        if (root->left != NULL)
            root->left->val = root->val;

        if (root->right != NULL)
            root->right->val = root->val;
    }

    change_children_sum(root->left);

    change_children_sum(root->right);

    int final_sum = 0;

    if (root->left != NULL)
        final_sum += root->left->val;

    if (root->right != NULL)
        final_sum += root->right->val;

    if (!is_leaf(root))
        root->val = final_sum;
}

29. BUILD TREE FROM PREORDER + INORDER

int find_index(
    int inorder_values[],
    int left,
    int right,
    int value)
{
    for (int index = left; index <= right; index++) {
        if (inorder_values[index] == value)
            return index;
    }

    return -1;
}

TreeNode *build_pre_in_helper(
    int preorder_values[],
    int *preorder_index,
    int inorder_values[],
    int left,
    int right)
{
    if (left > right)
        return NULL;

    int root_value =
        preorder_values[*preorder_index];

    (*preorder_index)++;

    TreeNode *root = create_node(root_value);

    int middle = find_index(
        inorder_values,
        left,
        right,
        root_value);

    root->left = build_pre_in_helper(
        preorder_values,
        preorder_index,
        inorder_values,
        left,
        middle - 1);

    root->right = build_pre_in_helper(
        preorder_values,
        preorder_index,
        inorder_values,
        middle + 1,
        right);

    return root;
}

TreeNode *build_pre_in(
    int preorder_values[],
    int inorder_values[],
    int size)
{
    int preorder_index = 0;

    return build_pre_in_helper(
        preorder_values,
        &preorder_index,
        inorder_values,
        0,
        size - 1);
}

30. BUILD TREE FROM POSTORDER + INORDER
============================================================

Important logic change:
Reading postorder backward gives Root -> Right -> Left.
Therefore build right first.

TreeNode *build_post_in_helper(
    int postorder_values[],
    int *postorder_index,
    int inorder_values[],
    int left,
    int right)
{
    if (left > right)
        return NULL;

    int root_value =
        postorder_values[*postorder_index];

    (*postorder_index)--;

    TreeNode *root = create_node(root_value);

    int middle = find_index(
        inorder_values,
        left,
        right,
        root_value);

    root->right = build_post_in_helper(
        postorder_values,
        postorder_index,
        inorder_values,
        middle + 1,
        right);

    root->left = build_post_in_helper(
        postorder_values,
        postorder_index,
        inorder_values,
        left,
        middle - 1);

    return root;
}

TreeNode *build_post_in(
    int postorder_values[],
    int inorder_values[],
    int size)
{
    int postorder_index = size - 1;

    return build_post_in_helper(
        postorder_values,
        &postorder_index,
        inorder_values,
        0,
        size - 1);
}

TreeNode *deserialize_helper(char **text)
{
    if (**text == '#') {
        *text = *text + 2;

        return NULL;
    }

    int sign = 1;
    int value = 0;

    if (**text == '-') {
        sign = -1;
        (*text)++;
    }

    while (**text != ',' && **text != '\0') {
        value = value * 10 + (**text - '0');
        (*text)++;
    }

    value = value * sign;

    if (**text == ',')
        (*text)++;

    TreeNode *root = create_node(value);

    root->left = deserialize_helper(text);

    root->right = deserialize_helper(text);

    return root;
}

TreeNode *deserialize_tree(char buffer[])
{
    char *text = buffer;

    return deserialize_helper(&text);
}

Is subtree BST?

typedef struct BSTInfo {
    int is_bst;
    int minimum;
    int maximum;
    int size;
} BSTInfo;

int largest_bst_answer = 0;

BSTInfo largest_bst_helper(TreeNode *root)
{
    BSTInfo result;

    if (root == NULL) {
        result.is_bst = 1;
        result.minimum = INT_MAX;
        result.maximum = INT_MIN;
        result.size = 0;
        return result;
    }

    BSTInfo left = largest_bst_helper(root->left);

    BSTInfo right = largest_bst_helper(root->right);

    int current_is_bst =
        left.is_bst
        && right.is_bst
        && root->val > left.maximum
        && root->val < right.minimum;

    if (current_is_bst) {
        result.is_bst = 1;
        result.size =
            1
            + left.size
            + right.size;

        result.minimum = root->val;

        if (left.minimum < result.minimum)
            result.minimum = left.minimum;

        result.maximum = root->val;

        if (right.maximum > result.maximum)
            result.maximum = right.maximum;

        if (result.size > largest_bst_answer)
            largest_bst_answer = result.size;

        return result;
    }

    result.is_bst = 0;
    result.minimum = INT_MIN;
    result.maximum = INT_MAX;
    result.size = 0;

    return result;
}

        TreeNode *node = stack[top--];

        printf("%d ", node->val);

        if (node->right != NULL)
            stack[++top] = node->right;

        if (node->left != NULL)
            stack[++top] = node->left;
    }
}

void iterative_inorder(TreeNode *root)
{

/* 45. MORRIS INORDER
 * No recursion.
 * No stack.
 * Temporarily connect predecessor to current node.
 */
void morris_inorder(TreeNode *root)
{
    TreeNode *current = root;

    while (current != NULL) {
        if (current->left == NULL) {
            printf("%d ", current->val);
            current = current->right;
        } else {
            TreeNode *predecessor = current->left;

            while (predecessor->right != NULL &&
                   predecessor->right != current) {
                predecessor = predecessor->right;
            }

            if (predecessor->right == NULL) {
                predecessor->right = current;
                current = current->left;
            } else {
                predecessor->right = NULL;
                printf("%d ", current->val);
                current = current->right;
            }
        }
    }
}
